cmake_minimum_required(VERSION 3.20)
project(MatrixRacing LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(option TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER GENERATE_DEBUG_SYMBOLS INTERPROCEDURAL_OPTIMIZATION ENABLE_ALL_WARNINGS OVERRIDE_CXX_FLAGS ENABLE_OBJECT_STREAM DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE USE_AVX2)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(../third_party/JoltPhysics/Build jolt)
find_package(Threads REQUIRED)
find_package(SDL2 2.0.18 REQUIRED CONFIG)
find_package(OpenGL REQUIRED)
find_package(ZLIB REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil libswscale)
add_library(nlohmann_json INTERFACE)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/nlohmann")
configure_file(../third_party/json.hpp "${CMAKE_CURRENT_BINARY_DIR}/include/nlohmann/json.hpp" COPYONLY)
target_include_directories(nlohmann_json INTERFACE "${CMAKE_CURRENT_BINARY_DIR}/include")
add_library(RacingCore STATIC src/World.cpp src/Physics.cpp src/RendererHud.cpp src/RendererGL.cpp
  src/HeadlessGL.cpp src/RaceRules.cpp src/RaceSession.cpp src/Pursuit.cpp src/CaptionStream.cpp src/WorldCaptions.cpp)
target_include_directories(RacingCore PUBLIC src)
target_link_libraries(RacingCore PUBLIC Jolt nlohmann_json SDL2::SDL2 OpenGL::GL ZLIB::ZLIB Threads::Threads)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
  target_compile_definitions(RacingCore PRIVATE CITYSAMPLE_HAVE_EGL=1)
  target_link_libraries(RacingCore PUBLIC OpenGL::EGL rt)
endif()
add_executable(CitySampleReferenceServer src/ReferenceServerMain.cpp)
target_link_libraries(CitySampleReferenceServer PRIVATE RacingCore)
add_executable(CitySampleBBoxGame src/MainSDL.cpp src/OfflineRecorder.cpp src/VehicleAudio.cpp
  src/LatencyDemo.cpp src/ReferenceDemo.cpp src/VideoEncoder.cpp src/ViewRecorder.cpp)
target_link_libraries(CitySampleBBoxGame PRIVATE RacingCore PkgConfig::AVCODEC)
set_target_properties(CitySampleReferenceServer CitySampleBBoxGame PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/bin")
include(CTest)
set(RACING_ASSET_ROOT "" CACHE PATH "Directory containing the racing assets folder")
if(BUILD_TESTING)
  foreach(name math minimap handling-menu keyboard-steering reference-stream)
    add_executable(test-${name} tools/${name}-tests.cpp)
    target_include_directories(test-${name} PRIVATE src)
    add_test(NAME racing-${name} COMMAND test-${name})
  endforeach()
  foreach(name caption local-race pursuit reference-presentation)
    add_executable(test-${name} tools/${name}-tests.cpp)
    target_link_libraries(test-${name} PRIVATE RacingCore)
    if(RACING_ASSET_ROOT)
      add_test(NAME racing-${name} COMMAND test-${name} "${RACING_ASSET_ROOT}")
    endif()
  endforeach()
endif()
