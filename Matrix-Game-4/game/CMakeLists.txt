cmake_minimum_required(VERSION 3.20)
project(MedievalCastleBBoxMap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
foreach(option TARGET_UNIT_TESTS TARGET_HELLO_WORLD TARGET_PERFORMANCE_TEST TARGET_SAMPLES TARGET_VIEWER GENERATE_DEBUG_SYMBOLS INTERPROCEDURAL_OPTIMIZATION ENABLE_ALL_WARNINGS OVERRIDE_CXX_FLAGS ENABLE_OBJECT_STREAM DEBUG_RENDERER_IN_DEBUG_AND_RELEASE PROFILER_IN_DEBUG_AND_RELEASE USE_AVX2 RECASTNAVIGATION_DEMO RECASTNAVIGATION_TESTS RECASTNAVIGATION_EXAMPLES)
  set(${option} OFF CACHE BOOL "" FORCE)
endforeach()
if(NOT WIN32)
  foreach(option USE_SSE4_1 USE_SSE4_2 USE_AVX USE_F16C USE_FMADD)
    set(${option} OFF CACHE BOOL "" FORCE)
  endforeach()
endif()
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
endif()
add_subdirectory(third_party/JoltPhysics/Build)
add_subdirectory(third_party/recastnavigation)
if(WIN32)
  add_executable(MedievalCastleBBoxMap WIN32 src/Main.cpp src/Renderer.cpp)
else()
  find_package(OpenGL REQUIRED)
  find_package(SDL2 REQUIRED)
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(CASTLE_AV REQUIRED IMPORTED_TARGET libavcodec libavformat libavutil)
  add_executable(MedievalCastleBBoxMap src/MainLinux.cpp src/RendererLinux.cpp src/HeadlessGL.cpp src/OfflineDemo.cpp src/DemoRecorder.cpp src/CombatAudio.cpp)
  target_include_directories(MedievalCastleBBoxMap PRIVATE third_party/DirectXMath/Inc)
  target_link_libraries(MedievalCastleBBoxMap PRIVATE SDL2::SDL2 OpenGL::GL Threads::Threads PkgConfig::CASTLE_AV)
  if(APPLE)
    target_compile_definitions(MedievalCastleBBoxMap PRIVATE GL_SILENCE_DEPRECATION)
  else()
    find_package(GLEW REQUIRED)
    target_link_libraries(MedievalCastleBBoxMap PRIVATE GLEW::GLEW)
    find_library(CASTLE_EGL_LIBRARY EGL)
    if(CASTLE_EGL_LIBRARY)
      target_compile_definitions(MedievalCastleBBoxMap PRIVATE CASTLE_HAVE_EGL)
      target_link_libraries(MedievalCastleBBoxMap PRIVATE ${CASTLE_EGL_LIBRARY})
    endif()
  endif()
endif()

if(NOT WIN32)
  enable_testing()
  add_executable(ExportCombatAudio tools/export-combat-audio.cpp src/CombatAudio.cpp)
  target_include_directories(ExportCombatAudio PRIVATE src)
  add_executable(CombatCueTests tools/test-combat-cues.cpp src/CombatAudio.cpp)
  target_include_directories(CombatCueTests PRIVATE src third_party third_party/DirectXMath/Inc)
  target_compile_options(CombatCueTests PRIVATE -UNDEBUG)
  add_test(NAME combat_cues COMMAND CombatCueTests)
  add_executable(SceneConfigTests tools/test-scene-config.cpp src/SceneConfig.cpp)
  target_include_directories(SceneConfigTests PRIVATE src)
  target_compile_options(SceneConfigTests PRIVATE -UNDEBUG)
  add_test(NAME scene_config COMMAND SceneConfigTests)
  add_executable(RoadSurfaceTests tools/test-road-surface.cpp src/RoadSurface.cpp)
  target_include_directories(RoadSurfaceTests PRIVATE src third_party third_party/DirectXMath/Inc)
  target_compile_options(RoadSurfaceTests PRIVATE -UNDEBUG)
  add_test(NAME road_surface COMMAND RoadSurfaceTests)
  add_executable(CameraInputTests tools/test-camera-input.cpp)
  add_executable(ReferencePacingTests tools/test-reference-pacing.cpp)
  target_compile_options(ReferencePacingTests PRIVATE -UNDEBUG)
  add_test(NAME reference_pacing COMMAND ReferencePacingTests)
  add_test(NAME camera_input COMMAND CameraInputTests)
  add_executable(DemoMediaTests tools/test-demo-media.cpp src/CombatAudio.cpp src/DemoRecorder.cpp)
  target_include_directories(DemoMediaTests PRIVATE src third_party)
  target_link_libraries(DemoMediaTests PRIVATE PkgConfig::CASTLE_AV Threads::Threads)
  target_compile_options(DemoMediaTests PRIVATE -UNDEBUG)
  target_compile_options(CameraInputTests PRIVATE -UNDEBUG)
  add_test(NAME demo_media COMMAND DemoMediaTests)
endif()
target_sources(MedievalCastleBBoxMap PRIVATE src/SceneConfig.cpp src/Map.cpp src/RoadSurface.cpp src/Collision.cpp src/Encounter.cpp src/Recording.cpp)
# CMake 3.24 can misdecode localized MSVC /showIncludes prefixes.
# Explicit native header dependencies prevent stale objects after layout changes.
file(GLOB native_headers CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.h" "${CMAKE_SOURCE_DIR}/src/*.inc")
set_source_files_properties(src/Main.cpp src/Map.cpp src/Renderer.cpp src/Collision.cpp src/Encounter.cpp src/Recording.cpp
  PROPERTIES OBJECT_DEPENDS "${native_headers}")
target_include_directories(MedievalCastleBBoxMap PRIVATE third_party)
if(WIN32)
  target_compile_definitions(MedievalCastleBBoxMap PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(MedievalCastleBBoxMap PRIVATE d3d11 dxgi d3dcompiler user32 gdi32 shell32 windowscodecs ole32 winmm)
endif()
if(MSVC)
  target_compile_options(MedievalCastleBBoxMap PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(MedievalCastleBBoxMap PRIVATE -Wall -Wextra -Wno-unused-parameter)
endif()
target_link_libraries(MedievalCastleBBoxMap PRIVATE Jolt Recast Detour)
set_target_properties(MedievalCastleBBoxMap PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")

if(NOT WIN32)
  add_executable(CastleReferenceServer src/ReferenceServerMain.cpp src/RendererLinux.cpp src/HeadlessGL.cpp
    src/SceneConfig.cpp src/Map.cpp src/RoadSurface.cpp src/Collision.cpp src/Encounter.cpp src/Recording.cpp)
  target_include_directories(CastleReferenceServer PRIVATE third_party third_party/DirectXMath/Inc)
  target_link_libraries(CastleReferenceServer PRIVATE Jolt Recast Detour SDL2::SDL2 OpenGL::GL Threads::Threads)
  if(APPLE)
    target_compile_definitions(CastleReferenceServer PRIVATE GL_SILENCE_DEPRECATION)
  else()
    target_link_libraries(CastleReferenceServer PRIVATE GLEW::GLEW)
    if(CASTLE_EGL_LIBRARY)
      target_compile_definitions(CastleReferenceServer PRIVATE CASTLE_HAVE_EGL)
      target_link_libraries(CastleReferenceServer PRIVATE ${CASTLE_EGL_LIBRARY})
    endif()
  endif()
  set_target_properties(CastleReferenceServer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
endif()
